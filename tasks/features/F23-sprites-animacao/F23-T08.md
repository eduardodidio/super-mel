```mermaid
  subgraph Assets
    ZIP[super_mel_ref_sprites.zip]

    SA[SpriteAnimator]
    ASM[AnimationStateMachine]
    EM[EffectManager]
  end

  subgraph Entities
    MEL[Mel.tsx]
    PROJ[Projectile.tsx]

  end

    HUD[HUD3D.tsx]

  MANIFEST --> SA
  SA --> MEL
  ASM --> MEL
  MEL --> EM
  EM --> EFF
  SA --> PROJ
  MANIFEST --> HUD
  HUD --> PORT
```
