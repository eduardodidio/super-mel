## User Story

As a developer, I need a versioned LevelData format with a generic entity array so that levels can persist coins, hearts, goals, checkpoints, and other objects beyond just blocks and a spawn point.

## Objective

Define the LevelDataV2 interface, EntityData types, and a migration function that auto-detects v1 data and produces valid v2.

### 2. Create `packages/shared/src/levelMigration.ts`

This file exports the migration function. It must be a pure function with no dependencies outside `types.ts`.

```typescript
import type { LevelDataV1, LevelDataV2, EntityData, BackgroundTheme } from "./types";

/**
 * Detect and migrate LevelData to v2 format.
 *
 * Detection logic:
 * - If `data.version === 2` -> already v2, return as-is (with defaults for missing optional fields)
 * - If `data.version` is undefined AND `data.spawnPoint` exists -> v1, migrate
 * - Otherwise -> throw Error("Unknown LevelData format")
 */
export function migrateLevelData(data: unknown): LevelDataV2 {
  if (!data || typeof data !== "object") {
    throw new Error("Invalid LevelData: not an object");
  }

  const obj = data as Record<string, unknown>;

  // Already v2
  if (obj.version === 2) {
    const v2 = obj as unknown as LevelDataV2;
    // Ensure entities array exists (defensive)
    if (!Array.isArray(v2.entities)) {
      v2.entities = [];
    }
    return v2;
  }

  // Detect v1: no version field, has spawnPoint
  if (obj.version === undefined && obj.spawnPoint && typeof obj.spawnPoint === "object") {
    const v1 = obj as unknown as LevelDataV1;
    const entities: EntityData[] = [];

    // Migrate spawnPoint to entity
    if (v1.spawnPoint) {
      entities.push({
        type: "spawn",
        x: v1.spawnPoint.x,
        y: v1.spawnPoint.y,
      });
    }

    return {
      version: 2,
      grid: v1.grid,
      width: v1.width,
      height: v1.height,
      entities,
      // theme not present in v1 -- leave undefined (caller uses level.background)
    };
  }

  throw new Error("Unknown LevelData format: cannot determine version");
}

/**
 * Type guard: check if data is already v2.
 */
export function isLevelDataV2(data: unknown): data is LevelDataV2 {
  return (
    typeof data === "object" &&
    data !== null &&
    (data as any).version === 2
  );
}
```
